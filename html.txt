    function markazApp() {
      return {
        currentTab: 'home',
        showAdmissionModal: false,
        showHifzModal: false,
        showReportModal: false,
        showCloudConfigModal: false,
        searchQuery: '',
        filterJuz: 'ALL',
        currentUser: null,
        selectedStudent: null,

        cloudConfig: {
          googleScriptUrl: localStorage.getItem('mf_script_url') || '',
          supabaseUrl: localStorage.getItem('mf_sp_url') || '',
          supabaseKey: localStorage.getItem('mf_sp_key') || ''
        },

        admissionForm: {
          fullName: '',
          phone: '',
          guardianName: '',
          track: 'HIFZ',
          priorJuz: 0,
          paymentChannel: 'TELEBIRR'
        },

        newLog: {
          studentCode: 'MF-2018-01',
          sabaqRange: '',
          sabqiJuz: 1,
          manzilJuz: 30,
          grade: 'MUMTAZ',
          notes: ''
        },

        students: [
          { id: 1, code: 'MF-2018-01', name: 'Abdulmalik Ahmed Mohammed', juzLevel: 28, guardianPhone: '0911223344' },
          { id: 2, code: 'MF-2018-02', name: 'Sumayya Bilal Siyum', juzLevel: 30, guardianPhone: '0922334455' },
          { id: 3, code: 'MF-2018-03', name: 'Hamza Kadir Ibrahim', juzLevel: 14, guardianPhone: '0933445566' },
          { id: 4, code: 'MF-2018-04', name: 'Zubayr Usman Ali', juzLevel: 7, guardianPhone: '0944556677' },
          { id: 5, code: 'MF-2018-05', name: 'Fathima Jibril Omar', juzLevel: 22, guardianPhone: '0955667788' }
        ],

        hifzLogs: [],

        get filteredStudents() {
          return this.students.filter(student => {
            const matchesQuery = student.name.toLowerCase().includes(this.searchQuery.toLowerCase()) || 
                                 student.code.toLowerCase().includes(this.searchQuery.toLowerCase());
            let matchesJuz = true;
            if (this.filterJuz === 'JUZ_1_10') matchesJuz = student.juzLevel <= 10;
            if (this.filterJuz === 'JUZ_11_20') matchesJuz = student.juzLevel > 10 && student.juzLevel <= 20;
            if (this.filterJuz === 'JUZ_21_30') matchesJuz = student.juzLevel > 20;
            return matchesQuery && matchesJuz;
          });
        },

        openAuthModal() {
          const email = prompt("Enter Mu'allim / Director Email or Google Account:", "director@markazulfurqan.edu.et");
          if (email) {
            this.currentUser = { name: email.split('@')[0], email: email };
            alert("Signed in as " + email + ". Access granted to academic management.");
          }
        },

        openStudentReport(student) {
          this.selectedStudent = student;
          this.showReportModal = true;
        },

        saveCloudConfig() {
          localStorage.setItem('mf_script_url', this.cloudConfig.googleScriptUrl);
          localStorage.setItem('mf_sp_url', this.cloudConfig.supabaseUrl);
          localStorage.setItem('mf_sp_key', this.cloudConfig.supabaseKey);
          this.showCloudConfigModal = false;
          alert("Cloud endpoints saved. Sync operational.");
        },

        async submitAdmissionForm() {
          const payload = {
            action: 'NEW_ADMISSION',
            ...this.admissionForm
          };

          // Optimistically add to local student roster
          const newStudent = {
            id: Date.now(),
            code: 'MF-2018-0' + (this.students.length + 1),
            name: this.admissionForm.fullName,
            juzLevel: parseInt(this.admissionForm.priorJuz) || 1,
            guardianPhone: this.admissionForm.phone
          };
          this.students.push(newStudent);

          // Send to Google Apps Script Webhook if configured
          if (this.cloudConfig.googleScriptUrl) {
            try {
              await fetch(this.cloudConfig.googleScriptUrl, {
                method: 'POST',
                mode: 'no-cors',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
              });
            } catch (err) {
              console.warn("Background cloud dispatch noted:", err);
            }
          }

          alert("Application submitted successfully for " + this.admissionForm.fullName + "!");
          this.showAdmissionModal = false;
          this.admissionForm = { fullName: '', phone: '', guardianName: '', track: 'HIFZ', priorJuz: 0, paymentChannel: 'TELEBIRR' };
        },

        async submitHifzLog() {
          const matchedStudent = this.students.find(s => s.code === this.newLog.studentCode);
          const payload = {
            action: 'LOG_HIFZ',
            studentName: matchedStudent ? matchedStudent.name : '',
            ...this.newLog
          };

          this.hifzLogs.push(payload);

          if (this.cloudConfig.googleScriptUrl) {
            try {
              await fetch(this.cloudConfig.googleScriptUrl, {
                method: 'POST',
                mode: 'no-cors',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
              });
            } catch (err) {
              console.warn("Background cloud dispatch noted:", err);
            }
          }

          alert("Sabaq & Daily Log stored for " + (matchedStudent ? matchedStudent.name : 'student') + ".");
          this.showHifzModal = false;
        },

        exportToCSV() {
          let csv = 'Student ID,Full Name,Current Juz,Guardian Phone\n';
          this.students.forEach(s => {
            csv += `"${s.code}","${s.name}","${s.juzLevel}","${s.guardianPhone}"\n`;
          });
          const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
          const link = document.createElement("a");
          const url = URL.createObjectURL(blob);
          link.setAttribute("href", url);
          link.setAttribute("download", `Markazul_Furqan_Students_${new Date().toISOString().slice(0,10)}.csv`);
          document.body.appendChild(link);
          link.click();
          document.body.removeChild(link);
        }
      }
    }
